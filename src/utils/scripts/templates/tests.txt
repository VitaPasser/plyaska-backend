import pytest
import pytest_asyncio
from httpx import AsyncClient

from src.models.{name} import {name}
from tests.test_utils.response_models import ResponseDo, ResponseModel


@pytest_asyncio.fixture
async def create_{lower_name}(client: AsyncClient) -> ResponseDo:
    async def _create_{lower_name}() -> ResponseModel:
        response = await client.post(
            url="/{lower_name}s/",
            content="""
                    {{}}
                    """,
        )
        model = {name}.model_validate(response.json())
        return ResponseModel(response=response, model=model)

    {lower_name} = await _create_{lower_name}()
    return ResponseDo(response={lower_name}.response, model={lower_name}.model, do=_create_{lower_name})


@pytest.mark.asyncio
async def test_create(client: AsyncClient, create_{lower_name}: ResponseDo):
    response = create_{lower_name}.response
    assert response.status_code == 201, response.json()


@pytest.mark.asyncio
async def test_find_all(client: AsyncClient, create_{lower_name}: ResponseDo):
    models_test = [(await create_{lower_name}.do()).model, (await create_{lower_name}.do()).model]
    response = await client.get(url="/{lower_name}s/")

    assert response.status_code == 200, response.json()

    models = [{name}.model_validate({lower_name}) for {lower_name} in response.json()]

    assert {{m.id for m in models}} >= {{m.id for m in models_test}}, response.json()


@pytest.mark.asyncio
async def test_find_by_id(client: AsyncClient, create_{lower_name}: ResponseDo):
    _id = create_{lower_name}.response.json()["_id"]
    response = await client.get(url=f"/{lower_name}s/{{_id}}")
    assert response.status_code == 200, response.json()


@pytest.mark.asyncio
async def test_update(client: AsyncClient, create_{lower_name}: ResponseDo):
    _id = (await create_{lower_name}.do()).model.id
    response = await client.patch(
        url=f"/{lower_name}s/{{_id}}",
        content="""
                {{}}
                """,
    )
    assert response.status_code == 200, response.json()
    model = {name}.model_validate(response.json())
    assert model.updated_at is not None and model.created_at is not None, response.json()


@pytest.mark.asyncio
async def test_delete(client: AsyncClient, create_{lower_name}: ResponseDo):
    _id = (await create_{lower_name}.do()).model.id
    response = await client.delete(url=f"/{lower_name}s/{{_id}}")
    assert response.status_code == 200, response.json()
